```rust
/// Only https links leave the app; the webview is not trusted to choose schemes or shells.
fn validate_external_url(raw: &str) -> Result<url::Url, String> {
    if raw.is_empty() || raw.trim() != raw || raw.chars().any(|c| c.is_control() || c.is_whitespace()) {
        return Err("Refusing to open a malformed URL".to_string());
    }
    let url = url::Url::parse(raw).map_err(|e| format!("Invalid URL: {e}"))?;
    if url.scheme() != "https" || url.host_str().map_or(true, str::is_empty) {
        return Err("Only https links can be opened".to_string());
    }
    Ok(url)
}

#[tauri::command]
pub async fn open_external_url(url: String) -> Result<(), String> {
    use std::process::Command;

    let url = validate_external_url(&url)?;
    // explorer.exe hands the URL to the default browser without cmd.exe parsing it.
    let result = if cfg!(target_os = "windows") {
        Command::new("explorer.exe").arg(url.as_str()).spawn().map(|_| ())
    } else if cfg!(target_os = "macos") {
        Command::new("open").arg(url.as_str()).spawn().map(|_| ())
    } else {
        Command::new("xdg-open").arg(url.as_str()).spawn().map(|_| ())
    };
    result.map_err(|e| format!("Failed to open URL: {}", e))
}
```

`explorer.exe` returns exit code 1 even when it succeeds, so use `spawn()` and don't inspect the exit status.

// frontend/src/lib/onboarding-transcription-model.ts
export const ONBOARDING_WHISPER_MODEL = 'large-v3-turbo';
export const ONBOARDING_WHISPER_SIZE_MB = 1549; // config.rs WHISPER_MODEL_CATALOG

export function isOnboardingModelEvent(payload: unknown): boolean {
  return (
    typeof payload === 'object' && payload !== null
    && (payload as { modelName?: unknown }).modelName === ONBOARDING_WHISPER_MODEL
  );
}

export function progressToDownloadedMb(progress: number): number {
  const pct = Math.min(100, Math.max(0, progress));
  return (pct / 100) * ONBOARDING_WHISPER_SIZE_MB;
}
